<script>
    // --- CONFIGURATION ---
    const CHANNEL_ID = 'YOUR_CHANNEL_ID'; // Replace with your ThingSpeak Channel ID
    const READ_API_KEY = 'YOUR_READ_API_KEY'; // Replace with your Read API Key
    const RESULTS_COUNT = 15; // Number of data points to display
    // ---------------------

    const url = `https://thingspeak.com{CHANNEL_ID}/feeds.json?api_key=${READ_API_KEY}&results=${RESULTS_COUNT}`;

    async function fetchTrackerData() {
        try {
            const response = await fetch(url);
            const data = await response.json();
            
            document.getElementById('loadingText').style.display = 'none';
            
            // Extract timestamps and field values (assuming Field 1 is your primary data)
            const labels = data.feeds.map(feed => {
                const date = new Date(feed.created_at);
                return date.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            });
            const fieldData = data.feeds.map(feed => parseFloat(feed.field1));
            const fieldName = data.channel.field1 || 'Sensor Data';

            // Render Chart
            const ctx = document.getElementById('iotChart').getContext('2d');
            new Chart(ctx, {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [{
                        label: fieldName,
                        data: fieldData,
                        borderColor: '#3498db',
                        backgroundColor: 'rgba(52, 152, 219, 0.1)',
                        borderWidth: 3,
                        fill: true,
                        tension: 0.3
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: { beginAtZero: false }
                    }
                }
            });
        } catch (error) {
            document.getElementById('loadingText').innerText = 'Error loading data. Check your Channel ID and API Key.';
            console.error('Error fetching data:', error);
        }
    }

    fetchTrackerData();
</script>
